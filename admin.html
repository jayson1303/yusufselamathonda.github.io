Kredensial Firebase SDK

Atur API Key dan konfigurasi project Firebase
Firestore Security Rules yang Direkomendasikan:
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /products/{document=**} { allow read: if true; allow write: if request.auth != null; }
    match /promos/{document=**} { allow read: if true; allow write: if request.auth != null; }
    match /testimonials/{document=**} { allow read: if true; allow write: if request.auth != null; }
    match /settings/{document=**} { allow read: if true; allow write: if request.auth != null; }
  }
}
